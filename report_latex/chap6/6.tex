\section{Kiến Trúc Decentralized Federated Learning}

\begin{figure}[H]
\centering
\includegraphics[width=0.9\textwidth]{image/ring_topology.png}
\caption{Sơ đồ kiến trúc Ring Topology trong Decentralized Federated Learning}
\end{figure}

\textbf{Đặc điểm kiến trúc DFL:}
\begin{itemize}
    \item \textbf{Giao tiếp ngang hàng:} Không có server trung tâm, các peers giao tiếp trực tiếp với nhau theo cấu trúc vòng tròn
    \item \textbf{Cấu trúc vòng tròn:} Mỗi peer chỉ kết nối với peer kế tiếp, tạo thành một vòng khép kín
    \item \textbf{Tổng hợp cục bộ:} Mỗi peer tự thực hiện tổng hợp với mô hình từ peer trước đó
    \item \textbf{Kiến trúc mô hình:} Autoencoder với cấu trúc Input(8) → Hidden(16) → Latent(4) → Hidden(16) → Output(8)
    \item \textbf{Cập nhật tuần tự:} Trọng số mô hình được truyền tuần tự từ Peer 0 → Peer 1 → ... → Peer 9 → Peer 0
\end{itemize}

\textbf{Quá trình huấn luyện trong DFL:}
\begin{enumerate}
    \item Mỗi peer huấn luyện mô hình trên dữ liệu cục bộ
    \item Peer nhận mô hình từ peer trước đó
    \item Thực hiện lấy trung bình: $w_{new} = \alpha \cdot w_{local} + (1-\alpha) \cdot w_{received}$
    \item Gửi mô hình đã tổng hợp đến peer tiếp theo
    \item Sau khi hoàn thành một vòng, bắt đầu vòng mới
\end{enumerate}

\textbf{Cấu hình:}
\begin{itemize}
    \item Số lượng Peers: 10
    \item Số vòng huấn luyện: 50
    \item Số epoch cục bộ: 1
    \item Tốc độ học: 0.001
    \item Kích thước batch: 128
    \item Thiết bị: CPU
\end{itemize}

\begin{enumerate}
    \item \textbf{DFL dữ liệu cân bằng:}
    \begin{itemize}

        \item Mỗi peer đóng góp đều nhau vào việc cập nhật mô hình
        \item Giảm thiên lệch từ các peers có nhiều dữ liệu
        \item Mất mát cuối cùng thấp nhất: 0.002392 (huấn luyện), 0.002425 (đánh giá)
    \end{itemize}
    
    \item \textbf{DFL dữ liệu không cân bằng:}
    \begin{itemize}
        \item Peer 0 (30\% dữ liệu) có ảnh hưởng lớn đến việc cập nhật mô hình
        \item Các peers nhỏ (như Peer 9 với 1\% dữ liệu) có tác động hạn chế
        \item Có thể gây quá khớp cho các mẫu dữ liệu của peers lớn
        \item Mất mát cuối cùng cao hơn: 0.002842 (huấn luyện), 0.002705 (đánh giá)
    \end{itemize}
\end{enumerate}

\textbf{So sánh với Centralized FL:}
\begin{itemize}
    \item DFL không cần central server, giảm single point of failure
    \item Privacy được bảo vệ tốt hơn nhờ P2P communication
    \item Training time có thể dài hơn do sequential updates
    \item Convergence phụ thuộc vào ring topology và data distribution
\end{itemize}

\textbf{2. Kiến trúc DFL phi tập trung:}
\begin{itemize}
    \item Cấu trúc ring hoạt động hiệu quả với 10 peers
    \item Truyền mô hình tuần tự đảm bảo chuyển giao kiến thức
    \item Không cần server trung tâm, tăng quyền riêng tư và tính bền vững
    \item Giao tiếp P2P giảm nghẽ cổ chai giao tiếp

    \item Bền vững với nhiều loại bất thường khác nhau

\textbf{4. Khả năng mở rộng và cân nhắc thực tế:}
\begin{itemize}
    \item Hệ thống mở rộng tốt với 10 peers, có thể mở rộng thêm

\textbf{Cho triển khai thực tế:}
\begin{enumerate}
    \item \textbf{Phân phối dữ liệu:} Ưu tiên phân phối cân bằng khi có thể
    \item \textbf{Lựa chọn ngưỡng:} Sử dụng phân vị 95 với hiệu chỉnh định kỳ
    \item \textbf{Giám sát:} Theo dõi hiệu suất của từng peer để phát hiện các peer chậm
    \item \textbf{Cập nhật mô hình:} Triển khai lưu điểm kiểm tra sau mỗi vòng
    \item \textbf{Khả năng chịu lỗi:} Xử lý lỗi peer với cơ chế hết thời gian chờ
\end{enumerate}

\textbf{Cho công việc tương lai:}
\begin{enumerate}
    \item Kiểm tra với cấu trúc mạng khác (lưới, phân cấp)
    \item Triển khai trọng số peer động dựa trên chất lượng dữ liệu
    \item Khám phá tốc độ học thích ứng cho từng peer
    \item Thêm mã hóa cho việc truyền trọng số mô hình
    \item So sánh hiệu suất với FL tập trung và các mô hình độc lập
\end{enumerate}

\subsection{Hạn chế}

\begin{itemize}
    \item Cập nhật tuần tự trong cấu trúc ring có thể chậm với nhiều peers
    \item Lỗi peer có thể phá vỡ vòng tròn và yêu cầu cơ chế khôi phục
    \item Không có cái nhìn tổng thể về tiến trình huấn luyện
    \item Dữ liệu không cân bằng vẫn ảnh hưởng đáng kể đến hiệu suất
    \item Ngưỡng cần được điều chỉnh cho từng lĩnh vực ứng dụng
\end{itemize}

\subsection{Kết luận}

Thực nghiệm cho thấy Decentralized Federated Learning với cấu trúc ring là một giải pháp khả thi cho phát hiện bất thường trong hệ thống giám sát vòng bi IoT. Phân phối dữ liệu cân bằng mang lại hiệu suất tốt nhất với mất mát đánh giá cuối cùng 0.002425 và 100% độ chính xác phát hiện bất thường. Kiến trúc DFL loại bỏ điểm lỗi đơn và tăng cường bảo vệ quyền riêng tư, phù hợp cho các ứng dụng IoT công nghiệp đòi hỏi bảo mật cao.
