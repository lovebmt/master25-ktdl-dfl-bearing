\section{Kiến Trúc Mô Hình Autoencoder}

Autoencoder được thiết kế để học representation của dữ liệu normal bearing:

\textbf{Kiến trúc:}
\begin{itemize}
    \item Lớp đầu vào: 8 đặc trưng
    \item Bộ mã hóa: $8 \rightarrow 4 \rightarrow 2$ (điểm thắt)
    \item Bộ giải mã: $2 \rightarrow 4 \rightarrow 8$
    \item Hàm kích hoạt: ReLU
    \item Hàm mất mát: Mean Squared Error (MSE)
\end{itemize}

\textbf{Cấu hình huấn luyện:}
\begin{itemize}
    \item Bộ tối ưu: Adam với tốc độ học $10^{-3}$
    \item Số epoch cục bộ: 1 epoch mỗi vòng
    \item Kích thước batch: 32
    \item Tổng số vòng: 50
\end{itemize}

\textbf{Nguyên lý:} Autoencoder học tái tạo dữ liệu bình thường. Khi gặp bất thường (hỏng vòng bi), sai số tái tạo sẽ cao hơn nhiều, cho phép phát hiện.

\textbf{MSE Formula và Computation:}

Loss function được sử dụng là Mean Squared Error (MSE):
$$MSE = \frac{1}{n} \sum_{i=1}^{n} (x_i - \hat{x}_i)^2$$

Trong đó:
\begin{itemize}
    \item $x_i$: giá trị gốc từ cảm biến thứ $i$
    \item $\hat{x}_i$: giá trị tái tạo từ autoencoder cho cảm biến thứ $i$
    \item $n$: số lượng features (8 sensors trong bearing dataset)
\end{itemize}

\textbf{Decision Rule cho Anomaly Detection:}
$$\text{Classification} = \begin{cases} 
\text{NORMAL} & \text{if } MSE < \text{Threshold} \\
\text{ANOMALY} & \text{if } MSE \geq \text{Threshold}
\end{cases}$$

\textbf{Tính toán Ngưỡng:}
$$\text{Threshold} = \text{Percentile}_{95}(\{MSE_i\}_{i=1}^{N})$$

Trong đó $N$ là tổng số mẫu kiểm tra trong tập kiểm định.

\section{Cấu Hình Decentralized Federated Learning}

Triển khai DFL thuần túy với kiến trúc P2P Ring Topology:

\begin{itemize}
    \item \textbf{Số peers:} 10 (mô phỏng 10 thiết bị IoT)
    \item \textbf{Cấu trúc mạng:} Vòng tròn - mỗi peer kết nối với 2 peer lân cận
    \item \textbf{Giao tiếp:} Peer-to-Peer (P2P) - không có server trung tâm
    \item \textbf{Tổng hợp:} Cục bộ tại mỗi peer (trung bình trọng số từ peer trước đó)
    \item \textbf{Tổng số vòng:} 50
    \item \textbf{Số epoch cục bộ:} 1 epoch mỗi vòng
    \item \textbf{Kích thước batch:} 128
    \item \textbf{Tốc độ học:} 0.001
    \item \textbf{Thiết bị:} CPU
\end{itemize}

\textbf{Luồng dữ liệu trong Ring Topology:} 
\begin{itemize}
    \item Peer 0 $\rightarrow$ Peer 1 $\rightarrow$ Peer 2 $\rightarrow$ ... $\rightarrow$ Peer 9 $\rightarrow$ Peer 0
    \item Mỗi peer huấn luyện mô hình cục bộ và gửi trọng số cho peer tiếp theo
    \item Peer nhận trọng số từ peer trước đó và thực hiện tổng hợp (lấy trung bình)
    \item Không có điểm lỗi đơn - hoàn toàn phi tập trung
\end{itemize}

\textbf{Ưu điểm của DFL P2P:} 
\begin{itemize}
    \item Tăng tính riêng tư: dữ liệu không bao giờ rời khỏi peer
    \item Giảm băng thông: chỉ giao tiếp với 2 peer lân cận
    \item Khả năng chịu lỗi: nếu một peer gặp sự cố, cấu trúc mạng có thể tái cấu hình
    \item Khả năng mở rộng: dễ dàng thêm/bớt peer mà không ảnh hưởng toàn hệ thống
\end{itemize}

\section{Quy Trình Thực Nghiệm}

\textbf{Thí nghiệm 1: Cân bằng (IID)}
\begin{enumerate}
    \item Chia đều 32,760 mẫu cho 10 peers (3,276 mẫu/peer)
    \item Khởi tạo mô hình với trọng số ngẫu nhiên
    \item Chạy 50 vòng DFL với cấu trúc P2P ring
    \item Mỗi peer huấn luyện mô hình cục bộ và trao đổi trọng số với peer kế tiếp
    \item Đánh giá trên tập kiểm tra cục bộ của mỗi peer
\end{enumerate}

\textbf{Thí nghiệm 2: Không cân bằng (Non-IID)}
\begin{enumerate}
    \item Phân phối không cân bằng theo luật lũy thừa (từ 329 đến 9,830 mẫu)
    \item Các bước còn lại giống Thí nghiệm 1
\end{enumerate}

\textbf{Chỉ số đánh giá:}
\begin{itemize}
    \item Mất mát huấn luyện: MSE trên tập huấn luyện của mỗi peer
    \item Mất mát đánh giá: MSE trên tập kiểm tra của mỗi peer
    \item Tốc độ hội tụ: số vòng để đạt mất mát < 0.005
    \item Giảm mất mát: chênh lệch giữa mất mát ban đầu và mất mát cuối cùng
    \item Ngưỡng phát hiện bất thường: phân vị 95 và Trung bình+2$\sigma$ của phân phối MSE
\end{itemize}
